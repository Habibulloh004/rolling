import sitemap, { generateSitemaps } from "../sitemap"; // Убедитесь, что оба импорта корректны

function generateSitemapXml(urls) {
  const urlset = urls
    .map((urlObj) => {
      return `
        <url>
          <loc>${urlObj.url}</loc>
          <lastmod>${urlObj.lastModified}</lastmod>
          <changefreq>daily</changefreq>
          <priority>0.8</priority>
        </url>
      `;
    })
    .join("\n");

  return `<?xml version="1.0" encoding="UTF-8"?>
  <urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">
    ${urlset}
  </urlset>`;
}

export default async function handler(req, res) {
  try {
    const sitemapData = await generateSitemaps();
    console.log(sitemapData)

    if (!Array.isArray(sitemapData) || sitemapData.length === 0) {
      console.error("Данные для sitemap не найдены");
      Response.status(500).send("Ошибка: данные для sitemap отсутствуют");
      return;
    }

    const urls = await Promise.all(
      sitemapData.map(async ({ type, id }) => {
        if (!type) {
          console.warn("Type отсутствует в одном из элементов sitemapData.");
          return [];
        }
        const result = await sitemap({ type, id });
        return result;
      })
    );

    const flatUrls = urls.flat().filter((urlObj) => urlObj && urlObj.url); // Фильтрация некорректных данных

    if (!flatUrls.length) {
      console.error("Нет корректных URL для генерации sitemap.");
      Response.json("Ошибка: URL отсутствуют");
      return;
    }

    const sitemapXml = generateSitemapXml(flatUrls);

    console.log("sss",sitemapXml)
    // Response.setHeader("Content-Type", "application/xml");
    Response.json(sitemapXml);
  } catch (error) {
    console.error("Ошибка генерации sitemap:", error);
    Response.json("Ошибка генерации sitemap");
  }
}
